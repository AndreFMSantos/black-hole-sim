# v3.1 (latest app, evolved copy of v0_6b-kerr-cuda): Kerr on CUDA. Linux with nvcc only (see the top-level CMakeLists.txt); links OpenGL via glfw for the window.
# v0.8c: OPTIONAL OptiX denoiser at build time. Only if the headers exist (code/third_party/optix-dev, outside Git, see
# tools/fetch_optix.sh); otherwise a stub is compiled and the rest stays the same. -DBHS_OPTIX=OFF turns it off even with the headers.
option(BHS_OPTIX "optional OptiX denoiser (needs code/third_party/optix-dev)" ON)
set(BHS_OPTIX_INCLUDE "${CMAKE_SOURCE_DIR}/third_party/optix-dev/include")
if(BHS_OPTIX AND EXISTS "${BHS_OPTIX_INCLUDE}/optix.h")
  set(BHS_DENOISE_SRC denoise_optix.cpp)
  set(BHS_HAVE_OPTIX ON)
  message(STATUS "v3_1-kerr-cuda-ui: denoiser OptiX ACTIVADO (${BHS_OPTIX_INCLUDE})")
else()
  set(BHS_DENOISE_SRC denoise_stub.cpp)
  set(BHS_HAVE_OPTIX OFF)
  message(STATUS "v3_1-kerr-cuda-ui: no OptiX (BHS_OPTIX=${BHS_OPTIX}; headers ${BHS_OPTIX_INCLUDE}/optix.h not found or turned off)")
endif()
# v3.1: interactive UI with Dear ImGui (MIT), optional: only if code/third_party/imgui exists (tools/fetch_imgui.sh); -DBHS_IMGUI=OFF turns it off
option(BHS_IMGUI "Dear ImGui interactive UI (needs code/third_party/imgui)" ON)
set(BHS_IMGUI_DIR "${CMAKE_SOURCE_DIR}/third_party/imgui")
set(BHS_UI_SRC "")
if(BHS_IMGUI AND EXISTS "${BHS_IMGUI_DIR}/imgui.h")
  set(BHS_UI_SRC ui.cpp ${BHS_IMGUI_DIR}/imgui.cpp ${BHS_IMGUI_DIR}/imgui_draw.cpp ${BHS_IMGUI_DIR}/imgui_tables.cpp ${BHS_IMGUI_DIR}/imgui_widgets.cpp
      ${BHS_IMGUI_DIR}/backends/imgui_impl_glfw.cpp ${BHS_IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
  set(BHS_HAVE_IMGUI ON)
  message(STATUS "v3_1-kerr-cuda-ui: ImGui UI ENABLED (${BHS_IMGUI_DIR})")
else()
  set(BHS_HAVE_IMGUI OFF)
  message(STATUS "v3_1-kerr-cuda-ui: no ImGui (using the bitmap HUD)")
endif()
add_executable(v3_1-kerr-cuda-ui main.cpp render.cu ${BHS_DENOISE_SRC} ${BHS_UI_SRC})
if(BHS_HAVE_IMGUI)
  target_include_directories(v3_1-kerr-cuda-ui SYSTEM PRIVATE "${BHS_IMGUI_DIR}" "${BHS_IMGUI_DIR}/backends")
  target_compile_definitions(v3_1-kerr-cuda-ui PRIVATE BHS_IMGUI=1)
  target_link_libraries(v3_1-kerr-cuda-ui PRIVATE ${CMAKE_DL_LIBS})
endif()
if(BHS_HAVE_OPTIX)
  target_include_directories(v3_1-kerr-cuda-ui PRIVATE "${BHS_OPTIX_INCLUDE}")
  target_compile_definitions(v3_1-kerr-cuda-ui PRIVATE BHS_HAVE_OPTIX=1)
  target_link_libraries(v3_1-kerr-cuda-ui PRIVATE ${CMAKE_DL_LIBS})
  if(TARGET CUDA::cupti)  # optional: --denoise-kernels lists the denoiser's CUDA kernels (CUPTI)
    target_link_libraries(v3_1-kerr-cuda-ui PRIVATE CUDA::cupti)
    target_compile_definitions(v3_1-kerr-cuda-ui PRIVATE BHS_HAVE_CUPTI=1)
  endif()
endif()
set_target_properties(v3_1-kerr-cuda-ui PROPERTIES CUDA_ARCHITECTURES "${BHS_CUDA_ARCH}" CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED ON)
target_link_libraries(v3_1-kerr-cuda-ui PRIVATE bhs_core bhs_exr glfw CUDA::cudart)
target_compile_options(v3_1-kerr-cuda-ui PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall -Wextra>
                                              $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr -lineinfo>)
target_compile_definitions(v3_1-kerr-cuda-ui PRIVATE BHS_SKY_DEFAULT="${CMAKE_CURRENT_SOURCE_DIR}/../../../assets/sky/starmap_2020_8k.exr")
target_compile_definitions(v3_1-kerr-cuda-ui PRIVATE BHS_SKY2_DEFAULT="${CMAKE_CURRENT_SOURCE_DIR}/../../../assets/sky/wise/wise_8k.exr")
